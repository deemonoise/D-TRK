> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task.
> **Do NOT commit per task.** The user commits everything at the end, after the hardware check
> (`git diff` / `git status` for review, not SHA ranges).

**Goal:** LFO destinations for almost every continuous synth parameter, plus LFO modulating another
LFO's rate, depth and retrig.

**Architecture:** New targets are appended to `enum class LfoDest` (file format unchanged: dest is a
byte, codecs clamp `>= Count` to 0). `Synth::control` sums LFO values per destination into an array
`la[LfoDest::Count]` and each consumer reads its slot. LFO-to-LFO uses the outputs/wraps of the
previous control tick (1 ms lag, any routing works incl. cycles). Retrig OFF (shared) LFOs get RATE /
RTRG mods only from shared sources, in `advanceInstLfos`. Design: `docs/plans/2026-10-09-lfo-dest-design.md`.

**Tech Stack:** C++17, PlatformIO, Unity tests on `native` (`pio test -e native -f <suite>`), firmware
`pio run -e wt32`.

**Destination order (append after DRIVE; values are file format, never reorder):**
`Reso, Fenv, Dly, Rvb, Bit, Srr, Fine, Duty, Sub, Noise, Semi2, Rate1, Rate2, Rate3, Rate4, Depth1,
Depth2, Depth3, Depth4, Rtrg1, Rtrg2, Rtrg3, Rtrg4, Count` → Count = 32.

**Scaling (l = -1..1 × depth/64, summed over LFOs):**
RESO, DLY, RVB, BIT, SRR, SUB, NOISE: `+64·l` on the 0..127 value (after its lock), clamped.
FENV: `+64·l` on fenv, clamp -64..63. FINE: `+l` semitones. DUTY: `+0.49·l`, clamp 0.01..0.99.
SEMI2: `+24·l` semitones on osc 2 only. RATE: target Hz × `2^(4·l)`. DEPTH: target output ×
`clamp(1 + Σl, 0, 2)`. RTRG: source phase wrap (crossed 1) → target phase 0 + new RND; self ignored.

**Validity per type:** DEC..CON — FM / DRUM / SYNTH. DUTY — CHIP. SUB / NOISE / SEMI2 — SYNTH. All
others — every type. (KIT and SAMPLE: as CHIP minus DUTY.)

```cpp
bool lfoDestValid(uint8_t dest, InstrType t) {
  if (dest >= static_cast<uint8_t>(LfoDest::Count)) return false;
  const auto d = static_cast<LfoDest>(dest);
  if (d >= LfoDest::Dec && d <= LfoDest::Con)
    return t == InstrType::Fm || t == InstrType::Drum || t == InstrType::Synth;
  if (d == LfoDest::Duty) return t == InstrType::Chip;
  if (d == LfoDest::Sub || d == LfoDest::Noise || d == LfoDest::Semi2) return t == InstrType::Synth;
  return true;
}

uint8_t lfoDestStep(uint8_t dest, int d, InstrType t, int self) {
  constexpr int kLast = static_cast<int>(LfoDest::Count) - 1;
  const int selfRtrg = self >= 0 && self < kLfos ? static_cast<int>(LfoDest::Rtrg1) + self : -1;
  auto ok = [&](int x) { return x != selfRtrg && lfoDestValid(static_cast<uint8_t>(x), t); };
  int v = dest > kLast ? 0 : dest;
  const int step = d > 0 ? 1 : -1;
  for (int i = 0; i < (d > 0 ? d : -d); ++i) {
    int nv = v + step;
    while (nv >= 0 && nv <= kLast && !ok(nv)) nv += step;
    if (nv < 0 || nv > kLast) break;
    v = nv;
  }
  return static_cast<uint8_t>(v);
}

const char* lfoDestName(uint8_t dest, InstrType t) {
  static const char* const kNames[] = {
      "PITCH", "DECAY", "COLOR", "SHAPE", "SWEEP", "CONTOUR", "VOL", "CUTOFF", "DRIVE",
      "RESO", "FENV", "DLY", "RVB", "BIT", "SRR", "FINE", "DUTY", "SUB", "NOISE", "SEMI2",
      "L1 RATE", "L2 RATE", "L3 RATE", "L4 RATE", "L1 DEPTH", "L2 DEPTH", "L3 DEPTH", "L4 DEPTH",
      "L1 RTRG", "L2 RTRG", "L3 RTRG", "L4 RTRG"};
  static const char* const kSyn[] = {"SHP1", "SHP2", "MIX", "DET", "SENV"};
  static_assert(sizeof(kNames) / sizeof(kNames[0]) == static_cast<int>(LfoDest::Count), "LFO dest names");
  if (dest >= static_cast<uint8_t>(LfoDest::Count)) dest = 0;
  const int mac = dest - static_cast<int>(LfoDest::Dec);
  if (t == InstrType::Synth && mac >= 0 && mac < kFmMacros) return kSyn[mac];
  return kNames[dest];
}
```

**Semantics.**
- Source i acts on target j when `lfoAt(m,i).depth != 0` and its dest is `RateJ / DepthJ / RtrgJ`.
  Source value `o_i` = the LFO's output of the previous control tick (-1..1 × depth/64, incl. its own
  DEPTH mod) — `v.lfoOut[i]` per voice, `instLfoOut_[instr][i]` shared.
- RATE: `hz_j *= exp2f(4 * Σ o_i)`. DEPTH: `out_j *= clampf(1 + Σ o_i, 0, 2)`. RTRG: if source i
  wrapped (phase crossed 1) on its last advance and i != j → target phase = 0, new RND, before the
  target advances this tick.
- Per-voice target (Retrig ON): any source. A shared source's wrap is `instLfoWrap_` bit i.
- Shared target (Retrig OFF): RATE / RTRG only from shared sources, computed in `advanceInstLfos`
  with `instLfoOut_` / `instLfoWrap_` of the previous tick; DEPTH applies per voice in `control`
  (any source) like for per-voice targets. `instLfoOut_` = shape × depth/64 × DEPTH mods from shared
  sources only.
- 0 depth source = off (nothing). `resetLfos` (note-on, not legato) clears `lfoOut` and `lfoWrap`.

```cpp
// LFO -> LFO modulation of an instrument's LFOs: per target j, rate octaves, depth factor and a
// restart. Sources: LFOs in srcMask with depth != 0 and a RateJ / DepthJ / RtrgJ target; out = their
// last outputs, wraps = bit i if source i wrapped.
struct LfoMods {
  float oct[kLfos] = {};
  float depth[kLfos] = {1, 1, 1, 1};
  uint8_t rtrg = 0;
};
MT_HOT LfoMods lfoMods(const Instrument& m, const float* out, uint8_t wraps, uint8_t srcMask) {
  LfoMods r;
  for (int i = 0; i < kLfos; ++i) {
    if (!(srcMask & (1u << i))) continue;
    const LfoCfg c = lfoAt(m, i);
    if (!c.depth) continue;
    const int d = c.dest;
    if (d >= ld(LfoDest::Rate1) && d < ld(LfoDest::Rate1) + kLfos) r.oct[d - ld(LfoDest::Rate1)] += 4.f * out[i];
    else if (d >= ld(LfoDest::Depth1) && d < ld(LfoDest::Depth1) + kLfos) r.depth[d - ld(LfoDest::Depth1)] += out[i];
    else if (d >= ld(LfoDest::Rtrg1) && d < ld(LfoDest::Rtrg1) + kLfos && d - ld(LfoDest::Rtrg1) != i &&
             (wraps & (1u << i)))
      r.rtrg |= static_cast<uint8_t>(1u << (d - ld(LfoDest::Rtrg1)));
  }
  for (float& x : r.depth) x = clampf(x, 0.f, 2.f);
  return r;
}
```

```cpp
MT_HOT float Synth::lfo(Voice& v, const LfoCfg& c, int i, int dt, const LfoMods& mods) {
  if (!c.depth) return v.lfoOut[i] = 0;
  float w;
  if (lfoFree(c.sync)) {
    w = lfoShape(c.wave, instLfoPhase_[v.instr][i], instLfoRnd_[v.instr][i]);
  } else {
    float& ph = v.lfoPhase[i];
    if (mods.rtrg & (1u << i)) {
      ph = 0;
      v.lfoRnd[i] = rnd();
    }
    float hz = lfoRateHz(c);
    if (mods.oct[i] != 0) hz *= exp2f(mods.oct[i]);
    ph += hz * dt * (1.f / kSynthRate);
    if (ph >= 1.f) {
      ph -= static_cast<int>(ph);
      v.lfoRnd[i] = rnd();
      v.lfoWrap |= static_cast<uint8_t>(1u << i);
    }
    w = lfoShape(c.wave, ph, v.lfoRnd[i]);
  }
  const int d = c.depth < -64 ? -64 : (c.depth > 63 ? 63 : c.depth);
  return v.lfoOut[i] = w * d * (1.f / 64.f) * mods.depth[i];
}
```
