**Goal:** Per-instrument drive and reverb send, a master compressor with a sidechain track, an arpeggiator with modes / rate / chord cycling, velocity → cutoff / DECAY. Old projects and presets sound unchanged.

**Architecture:** Three hardware-free DSP units in `lib/core` (`synth_drive.h`, `synth_reverb.*`, `synth_comp.*`) hooked into `Synth::render`: drive per voice before the filter; reverb from an `rsend_` bus like the delay; the compressor on the mix after the returns, keyed by the mix plus an `sc_` bus of the sidechain track's voices. New lock bits (`uint16_t lockMask`) and fx `DRV`, `RVB`, `ARM`; ARP state in `TrackRt` grows a mode / rate and a chord note list. New bytes go into reserved slots of the `FMIN` record and appended `AUDI` bytes.

**Step 3: Implement** `synth_drive.h`:
```cpp
#pragma once
#include <math.h>
#include <stdint.h>
namespace mt {
// tanh waveshaper: y = tanh(g x) / tanh(g), g = 1..8 from drive 0..127; drive 0 = bypass (bit-exact).
// 257-point table over [-8, 8], linear interpolation; no transcendental per sample.
class Drive {
 public:
  static constexpr int kTab = 257;
  void set(uint8_t drive) {  // per control update
    g_ = drive ? 1.f + (drive > 127 ? 127 : drive) * (7.f / 127.f) : 0.f;
    norm_ = drive ? 1.f / tanhf(g_) : 1.f;
  }
  bool on() const { return g_ != 0.f; }
  float process(float x) const {
    if (!on()) return x;
    float u = x * g_;
    u = u < -8.f ? -8.f : (u > 8.f ? 8.f : u);
    const float p = (u + 8.f) * 16.f;         // table position 0..256
    const int i = static_cast<int>(p);
    const float f = p - i;
    const float* t = table();
    const float a = t[i], b = t[i + 1 < kTab ? i + 1 : kTab - 1];
    return (a + (b - a) * f) * norm_;
  }
 private:
  static const float* table() {
    static float t[kTab];
    static bool init = false;
    if (!init) { for (int i = 0; i < kTab; ++i) t[i] = tanhf(-8.f + i * (1.f / 16.f)); init = true; }
    return t;
  }
  float g_ = 0, norm_ = 1;
};
}  // namespace mt
```
(The table is built on first use from the audio task — 257 `tanhf` once, ~50 µs; acceptable. If the first block must be clean, call `Drive().set(1)` in `Synth`'s constructor.)

```cpp
#pragma once
#include <stdint.h>
namespace mt {
// Freeverb-lite mono reverb: 4 parallel damped feedback combs, 2 series allpasses. Float buffer
// (kBufLen floats) handed in by the caller (PSRAM on the device); no buffer = silent.
class Reverb {
 public:
  static constexpr int kCombs = 4, kAllpasses = 2;
  static constexpr int kCombLen[kCombs] = {1116, 1188, 1277, 1356};
  static constexpr int kApLen[kAllpasses] = {556, 441};
  static constexpr int kBufLen = 1116 + 1188 + 1277 + 1356 + 556 + 441;  // 5934
  void setBuffer(float* buf, int len);  // len must be >= kBufLen, else off
  void clear();
  // mix[i] += wet * level / 127 * 0.5. size / damp / level 0..127.
  void process(const float* in, float* mix, int n, uint8_t size, uint8_t damp, uint8_t level);
 private:
  float* buf_ = nullptr;
  float* comb_[kCombs] = {};
  float* ap_[kAllpasses] = {};
  int combIdx_[kCombs] = {}, apIdx_[kAllpasses] = {};
  float store_[kCombs] = {};
  int combLen_[kCombs] = {};  // current (size-scaled) lengths
};
}
```
`synth_reverb.cpp`:
```cpp
void Reverb::setBuffer(float* buf, int len) {
  buf_ = len >= kBufLen ? buf : nullptr;
  if (!buf_) return;
  float* p = buf_;
  for (int c = 0; c < kCombs; ++c) { comb_[c] = p; p += kCombLen[c]; }
  for (int a = 0; a < kAllpasses; ++a) { ap_[a] = p; p += kApLen[a]; }
  clear();
}
void Reverb::clear() {
  if (!buf_) return;
  for (int i = 0; i < kBufLen; ++i) buf_[i] = 0;
  for (int c = 0; c < kCombs; ++c) { combIdx_[c] = 0; store_[c] = 0; combLen_[c] = kCombLen[c]; }
  for (int a = 0; a < kAllpasses; ++a) apIdx_[a] = 0;
}
void Reverb::process(const float* in, float* mix, int n, uint8_t size, uint8_t damp, uint8_t level) {
  if (!buf_ || level == 0) return;
  const float sz = (size > 127 ? 127 : size) * (1.f / 127.f);
  const float fb = 0.70f + 0.28f * sz;
  const float d = (damp > 127 ? 127 : damp) * (0.4f / 127.f);
  const float wet = (level > 127 ? 127 : level) * (0.5f / 127.f);
  for (int c = 0; c < kCombs; ++c) {
    const int len = static_cast<int>(kCombLen[c] * (0.5f + 0.5f * sz));
    combLen_[c] = len < 1 ? 1 : len;
    if (combIdx_[c] >= combLen_[c]) combIdx_[c] = 0;
  }
  for (int i = 0; i < n; ++i) {
    const float x = in[i] * 0.25f;  // comb input gain (4 combs sum)
    float acc = 0;
    for (int c = 0; c < kCombs; ++c) {
      float* line = comb_[c];
      const float out = line[combIdx_[c]];
      store_[c] = out * (1.f - d) + store_[c] * d;
      line[combIdx_[c]] = x + store_[c] * fb;
      if (++combIdx_[c] >= combLen_[c]) combIdx_[c] = 0;
      acc += out;
    }
    for (int a = 0; a < kAllpasses; ++a) {
      float* line = ap_[a];
      const float b = line[apIdx_[a]];
      line[apIdx_[a]] = acc + b * 0.5f;
      acc = b - acc * 0.5f;  // allpass, g = 0.5 (Freeverb form)
      if (++apIdx_[a] >= kApLen[a]) apIdx_[a] = 0;
    }
    mix[i] += acc * wet;
  }
}
```

```cpp
#pragma once
#include <math.h>
#include <stdint.h>
namespace mt {
// Master compressor: peak detector (1 ms attack, 20..1000 ms release), 4:1 above a threshold set by
// amt (-6 .. -30 dB), makeup = half the threshold drop. Gain computed every kHold samples, applied per
// sample. Key = mix, plus the sidechain bus weighted by scDepth. amt 0 = bypass (bit-exact).
class Compressor {
 public:
  static constexpr int kHold = 8;
  static constexpr float kRate = 32000.f;
  void reset() { env_ = 0; gain_ = 1; }
  void process(float* mix, const float* sc, int n, uint8_t amt, uint8_t rel, uint8_t scDepth) {
    if (amt == 0) return;
    const float a = amt > 127 ? 127 : amt;
    const float thrDb = -6.f - 24.f * a / 127.f;
    const float makeup = exp2f((-thrDb) * 0.5f / 6.02f);
    const float relMs = 20.f * powf(50.f, (rel > 127 ? 127 : rel) / 127.f);
    const float ca = 1.f - expf(-1.f / (0.001f * kRate));
    const float cr = 1.f - expf(-1.f / (relMs * 0.001f * kRate));
    const float scW = (scDepth > 127 ? 127 : scDepth) * (4.f / 127.f);
    for (int i = 0; i < n; ++i) {
      float key = fabsf(mix[i]);
      if (sc) key += fabsf(sc[i]) * scW;
      env_ += (key > env_ ? ca : cr) * (key - env_);
      if (i % kHold == 0) {
        const float envDb = env_ > 1e-6f ? 6.02f * log2f(env_) : -120.f;
        const float over = envDb - thrDb;
        gain_ = (over > 0 ? exp2f(-over * 0.75f / 6.02f) : 1.f) * makeup;
      }
      mix[i] *= gain_;
    }
  }
 private:
  float env_ = 0, gain_ = 1;
};
}
```
